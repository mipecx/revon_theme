// Package main is a short demo of the Revon theme.
package main

import (
    "context"
    "fmt"
    "strings"
)

const MaxWorkers = 8
const greeting = "Hello, Revon!"

// Vacancy is a job posting.
type Vacancy struct {
    ID     int
    Title  string
    Remote bool
    Salary float64
    Tags   []string
}

type Fetcher interface {
    Fetch(ctx context.Context, q string) ([]Vacancy, error)
}

func (v Vacancy) String() string {
    remote := "office"
    if v.Remote {
        remote = "remote"
    }
    return fmt.Sprintf("%s (%s) — %.1f", v.Title, remote, v.Salary)
}

func Filter(vs []Vacancy, pred func(Vacancy) bool) []Vacancy {
    out := make([]Vacancy, 0, len(vs))
    for _, v := range vs {
        if pred(v) {
            out = append(out, v)
        }
    }
    return out
}

func main() {
    vacancies := []Vacancy{
        {ID: 1, Title: "Go Developer", Remote: true, Salary: 150000.5, Tags: []string{"go", "kubernetes"}},
        {ID: 2, Title: "SRE", Remote: false, Salary: 180000.0, Tags: []string{"linux", "terraform"}},
    }

    remote := Filter(vacancies, func(v Vacancy) bool {
        return v.Remote && v.Salary > 100000
    })

    for i, v := range remote {
        fmt.Printf("%d. %s\n", i+1, strings.ToUpper(v.String()))
    }
}
